1. `app/actions/trees.ts` — RESEARCH.md §7 verbatim:

```ts
'use server';

import { getUserIdOrThrow } from '@/lib/auth';
import { supabaseServer } from '@/lib/supabase/server';

export type TreeListItem = {
  id: string;
  name: string;
  ownerUserId: string;
  role: 'owner' | 'editor' | 'viewer';
  memberCount: number;
  updatedAt: string;
};

export async function listMyTrees(): Promise<TreeListItem[]> {
  const userId = await getUserIdOrThrow();
  const supabase = await supabaseServer();
  const { data, error } = await supabase
    .from('tree_members')
    .select(`
      role,
      tree:trees!inner(
        id,
        name,
        owner_user_id,
        updated_at
      )
    `)
    .eq('user_id', userId)
    .eq('status', 'active')
    .order('updated_at', { foreignTable: 'trees', ascending: false });

  if (error) throw new Error(`listMyTrees failed: ${error.message}`);
  if (!data) return [];

  return data.map((row: any) => ({
    id: row.tree.id,
    name: row.tree.name,
    ownerUserId: row.tree.owner_user_id,
    role: row.role as 'owner' | 'editor' | 'viewer',
    memberCount: 0, // Phase 5 adds aggregation
    updatedAt: row.tree.updated_at,
  }));
}

export async function createNewTree(name = 'Untitled tree'): Promise<string> {
  const userId = await getUserIdOrThrow();
  const supabase = await supabaseServer();
  const { data, error } = await supabase.rpc('bootstrap_tree', {
    p_owner_user_id: userId,
    p_tree_name: name,
    p_seed_person_name: 'You',
  });
  if (error || !data) throw new Error(`createNewTree failed: ${error?.message ?? 'unknown'}`);
  return data as string;
}

export async function renameTree(treeId: string, name: string): Promise<void> {
  await getUserIdOrThrow();
  const trimmed = name.trim().slice(0, 80);
  if (trimmed.length === 0) return; // Silent revert per UI-SPEC §Inline tree rename
  const supabase = await supabaseServer();
  const { error } = await supabase.from('trees').update({ name: trimmed }).eq('id', treeId);
  if (error) throw new Error(`renameTree failed: ${error.message}`);
}
```

Critical:
- `'use server';` first line
- `getUserIdOrThrow()` called FIRST in every action (Pitfall 1-6)
- `renameTree` MUST trim + slice(0, 80) + early-return silently when empty — UI-SPEC §Inline tree rename "Empty name: Revert to previous name; no error shown"
- `createNewTree` defaults to `'Untitled tree'` (the topbar's TreeSwitcher does NOT pass a name; user renames inline after navigation)

2. `lib/utils/hashUserId.ts` — deterministic avatar color picker:

```ts
const PALETTE = [
  'oklch(0.52 0.14 250)', // accent indigo
  'oklch(0.62 0.13 150)', // success green
  'oklch(0.72 0.14 75)',  // warning gold
  'oklch(0.70 0.10 300)', // lilac (UI-SPEC Open Q #3 4th color)
] as const;

// Simple deterministic hash: sum char codes
export function hashUserIdToColor(userId: string): string {
  let h = 0;
  for (let i = 0; i < userId.length; i++) h = (h * 31 + userId.charCodeAt(i)) | 0;
  const idx = Math.abs(h) % PALETTE.length;
  return PALETTE[idx];
}

export function initialsFromName(name: string): string {
  const parts = name.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return '??';
  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
}
```

```tsx
'use client';

import { useRef, useState, useTransition, KeyboardEvent, ChangeEvent } from 'react';
import { renameTree } from '@/app/actions/trees';

export type TreeTitleProps = {
  treeId: string;
  name: string;
  peopleCount: number;
};

export default function TreeTitle({ treeId, name, peopleCount }: TreeTitleProps) {
  const [isEditing, setIsEditing] = useState(false);
  const [draft, setDraft] = useState(name);
  const [displayName, setDisplayName] = useState(name);
  const inputRef = useRef<HTMLInputElement>(null);
  const [, startTransition] = useTransition();

  const enterEdit = () => {
    setDraft(displayName);
    setIsEditing(true);
    // Autofocus + select-all after paint
    queueMicrotask(() => {
      inputRef.current?.focus();
      inputRef.current?.select();
    });
  };

  const commit = () => {
    const trimmed = draft.trim().slice(0, 80);
    if (trimmed.length === 0) {
      // Silent revert per UI-SPEC §Inline tree rename
      setIsEditing(false);
      setDraft(displayName);
      return;
    }
    setDisplayName(trimmed); // optimistic
    setIsEditing(false);
    startTransition(async () => {
      try {
        await renameTree(treeId, trimmed);
      } catch {
        // Silent revert on server error per UI-SPEC (no toast infra in Phase 1)
        setDisplayName(name);
      }
    });
  };

  const cancel = () => {
    setIsEditing(false);
    setDraft(displayName);
  };

  const onKey = (e: KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Enter') { e.preventDefault(); commit(); }
    if (e.key === 'Escape') { e.preventDefault(); cancel(); }
  };

  if (isEditing) {
    return (
      <input
        ref={inputRef}
        className="w-full border border-rule bg-bg-card text-[14px] text-ink focus:border-accent focus:outline-none"
        style={{ padding: '8px 12px' }}
        value={draft}
        maxLength={80}
        placeholder="Name your tree"
        onChange={(e: ChangeEvent<HTMLInputElement>) => setDraft(e.target.value)}
        onKeyDown={onKey}
        onBlur={commit}
        aria-label="Rename tree"
      />
    );
  }

  const meta = peopleCount === 1 ? '· 1 person' : `· ${peopleCount} people`;

  return (
    <button
      type="button"
      onClick={enterEdit}
      title="Click to rename"
      className="flex items-baseline gap-[8px] text-left hover:bg-bg-soft transition-colors"
      style={{ padding: '4px 8px' }}
    >
      <span className="text-[13px] font-semibold tracking-[-0.005em] text-ink">{displayName}</span>
      <span className="font-mono text-[11px] text-ink-3">{meta}</span>
    </button>
  );
}
```

2. `app/(app)/tree/[treeId]/page.tsx` — the ACTUAL tree route. RSC fetches the tree + people via `supabaseServer()` (RLS-gated), renders `<TopBar>` + `<GridBackground>` + `<SeedPersonNode>` + `<EmptyTreeOverlay>`:

```tsx
import { notFound } from 'next/navigation';
import { getUserIdOrThrow, getUserProfile } from '@/lib/auth';
import { supabaseServer } from '@/lib/supabase/server';
import TopBar from '@/components/shell/TopBar';
import GridBackground from '@/components/shell/GridBackground';
import SeedPersonNode from '@/components/shell/SeedPersonNode';
import EmptyTreeOverlay from '@/components/shell/EmptyTreeOverlay';
import AuthError from '@/components/shell/AuthError';

export const dynamic = 'force-dynamic';

type PageParams = { params: Promise<{ treeId: string }> };

export default async function TreePage({ params }: PageParams) {
  const { treeId } = await params;
  await getUserIdOrThrow();
  const profile = await getUserProfile();
  const supabase = await supabaseServer();

  // RLS enforces access — if the tree isn't visible to this user, treeRes.data is null
  const { data: tree, error: treeErr } = await supabase
    .from('trees')
    .select('id, name, owner_user_id')
    .eq('id', treeId)
    .maybeSingle();

  if (treeErr) throw new Error(`Load tree failed: ${treeErr.message}`);
  if (!tree) {
    // RLS blocked or the tree doesn't exist — show the "This tree isn't yours to view" fallback
    return <AuthError variant="rls-reject" />;
  }

  const { data: people, error: peopleErr } = await supabase
    .from('people')
    .select('id, name, x, y, is_me')
    .eq('tree_id', treeId);

  if (peopleErr) throw new Error(`Load people failed: ${peopleErr.message}`);
  const peopleList = people ?? [];
  const seed = peopleList.find((p) => p.is_me) ?? peopleList[0];

  return (
    <>
      <TopBar
        currentTreeId={tree.id}
        treeName={tree.name}
        peopleCount={peopleList.length}
        userId={profile?.id ?? ''}
        displayName={profile?.displayName ?? 'You'}
        email={profile?.email ?? ''}
      />

      {/* Canvas region (Phase 1 = static 1-node; Phase 2 adds pan/zoom) */}
      <section
        aria-label="Family tree canvas"
        className="relative"
        style={{ minHeight: 'calc(100vh - 52px)' }}
        tabIndex={0}
      >
        <GridBackground />
        {seed && (
          <div className="absolute" style={{ left: '50%', top: 180, transform: 'translate(-50%, 0)' }}>
            <SeedPersonNode name={seed.name} x={0} y={0} />
          </div>
        )}
        {peopleList.length <= 1 && <EmptyTreeOverlay />}
      </section>
    </>
  );
}
```

Critical:
- `dynamic = 'force-dynamic'` — RSC reads auth, must run per-request
- `params: Promise<...>` — Next 15+ params are async-awaited
- `getUserIdOrThrow()` as defense-in-depth (Pitfall 1-6)
- Use `.maybeSingle()` on the tree query — RLS returns 0 rows for unauthorized access, we treat null as "RLS rejected" and render `<AuthError variant="rls-reject" />`
- The seed node is centered horizontally via `left: 50%` + `translate(-50%, 0)` — Phase 1 doesn't have pan/zoom, so this manual centering keeps the seed visible
- `<EmptyTreeOverlay />` shows while `peopleList.length <= 1` (the seed alone counts as empty)
- `tabIndex={0}` on canvas section so the keyboard Tab order (UI-SPEC Accessibility) can include the canvas as a focusable region
- Canvas minHeight fills viewport below the 52px topbar

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser -> `app/(app)/layout.tsx` auth guard | Unauthenticated request must be redirected; middleware catches most but server re-checks as defense-in-depth |
| browser -> `app/(app)/tree/[treeId]/page.tsx` | Tree id comes from URL — untrusted; RLS is the authz boundary, code treats nullish tree as "rejected" |
| client component -> server action (`listMyTrees`, `createNewTree`, `renameTree`) | Client-initiated server calls — every action re-checks `auth()` (Pitfall 1-6) |
| user menu Sign out click -> Clerk `signOut()` | Client-side Clerk SDK clears httpOnly cookie; server doesn't need to participate |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Spoofing | User visits `/tree/<someoneElseTreeId>` via direct URL | mitigate | RSC RLS-gated SELECT returns null; page renders `<AuthError variant="rls-reject" />` with UI-SPEC copy `This tree isn't yours to view.` |
| T-04-02 | Tampering | User posts a payload to `renameTree` claiming a treeId they don't own | mitigate | `renameTree` uses Supabase client with RLS; UPDATE on `trees` where `id = treeId` fails the `trees_update_if_owner_or_editor` policy (plan 02) — no data mutation occurs |
| T-04-03 | Tampering | User calls `createNewTree` with a name that's 1MB of garbage | mitigate | Server action trims + relies on Postgres text column; `trees.name` is unbounded in Phase 1 schema but HTTP body size limits + lack of index-based attack mean the exposure is limited (document as "consider adding `check (length(name) <= 200)` in a v1.1 patch if abused") |
| T-04-04 | Information Disclosure | User menu `title={displayName} · {email}` leaks PII to screen recording | accept | Email/name are the user's own info; tooltip is UI-SPEC-mandated |
| T-04-05 | Information Disclosure | Server action `listMyTrees` returns other users' tree data | mitigate | RLS gates SELECT on `tree_members` — only the caller's memberships return; joined `trees` rows are additionally gated by `trees_select_if_accessible` |

| T-04-08 | Repudiation | No server-side log of rename/create actions | accept | Supabase `updated_at` trigger records last-edit time; full audit log is out of scope for v1 |
| T-04-09 | Spoofing | Attacker intercepts `signOut()` redirect to phishing page | accept | Redirect URL is hardcoded to `/sign-in` (same-origin); Clerk SDK validates redirect against allowed origins configured in Clerk Dashboard |
