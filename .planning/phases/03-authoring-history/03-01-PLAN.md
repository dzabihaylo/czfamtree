**lib/graph/placement.ts** — verbatim from RESEARCH §"Code Examples §D":
```typescript
import { NODE_W, NODE_H } from './edges';

const COLLISION_RADIUS = NODE_W + 32; // 212px (D-13)
const NUDGE_STEP = NODE_H + 16;       // 92px (D-13)
const MAX_ITER = 20;                  // bounded fallback (D-13, Pitfall #9)

export type Pt = { x: number; y: number };
export type RelativeKind = 'parent' | 'spouse' | 'child' | 'sibling';

/**
 * Returns a position clear of any same-row collision in `people`, by
 * stepping +92 along Y up to 20 times. After 20 iterations it returns
 * the last attempted position rather than throwing — pathological tree
 * shapes (Pitfall #9) land 1840px below the anchor; user can drag.
 */
export function nudgePosition(
  initial: Pt,
  people: Record<string, { x: number; y: number }>,
): Pt {
  let { x, y } = initial;
  for (let i = 0; i < MAX_ITER; i++) {
    const collides = Object.values(people).some(
      (p) => Math.abs(p.x - x) < COLLISION_RADIUS && Math.abs(p.y - y) < NODE_H,
    );
    if (!collides) return { x, y };
    y += NUDGE_STEP;
  }
  return { x, y };
}

export function initialOffsetFor(kind: RelativeKind, anchor: Pt): Pt {
  switch (kind) {
    case 'spouse':  return { x: anchor.x + NODE_W + 32, y: anchor.y };
    case 'parent':  return { x: anchor.x, y: anchor.y - NODE_H - 80 };
    case 'child':   return { x: anchor.x, y: anchor.y + NODE_H + 80 };
    case 'sibling': return { x: anchor.x - NODE_W - 32, y: anchor.y };
  }
}
```

```typescript
import type { Person } from '@/lib/store/tree-store';

export type RelationKind = 'self' | 'parent' | 'spouse' | 'child' | 'sibling' | 'relative';
export type RelationHint = { kind: RelationKind; qualifier?: string };

const MAX_HOPS = 4;

/**
 * BFS-walks the relationship graph from `fromId` (typically the is_me
 * person) and returns a relation hint kind for `toId`. Used by SRCH-02
 * result rows; format is `{KIND} · {QUALIFIER}` per UI-SPEC §5.
 */
export function relationFrom(
  people: Record<string, Person>,
  fromId: string,
  toId: string,
): RelationHint {
  if (fromId === toId) return { kind: 'self', qualifier: 'YOU' };
  const from = people[fromId];
  const to = people[toId];
  if (!from || !to) return { kind: 'relative' };

  if (from.spouseIds.includes(toId)) return { kind: 'spouse', qualifier: 'OF YOU' };
  if (from.parentIds.includes(toId)) return { kind: 'parent', qualifier: 'OF YOU' };
  if (from.childIds.includes(toId))  return { kind: 'child',  qualifier: 'OF YOU' };
  if (from.parentIds.some((pid) => to.parentIds.includes(pid))) {
    return { kind: 'sibling', qualifier: 'OF YOU' };
  }

  // BFS up to MAX_HOPS for grandparent / in-law / cousin chains.
  // For v1, any non-direct, non-sibling tie returns 'relative' without
  // a qualifier — matches UI-SPEC §"Search Palette" copy `RELATIVE`.
  // Walk via spouseIds/parentIds/childIds; cap visited nodes at MAX_HOPS depth.
  const visited = new Set<string>([fromId]);
  let frontier: string[] = [fromId];
  for (let hop = 0; hop < MAX_HOPS; hop++) {
    const next: string[] = [];
    for (const id of frontier) {
      const node = people[id];
      if (!node) continue;
      for (const nbr of [...node.spouseIds, ...node.parentIds, ...node.childIds]) {
        if (visited.has(nbr)) continue;
        if (nbr === toId) return { kind: 'relative' };
        visited.add(nbr);
        next.push(nbr);
      }
    }
    frontier = next;
    if (frontier.length === 0) break;
  }
  return { kind: 'relative' };
}
```

```sql
-- Phase 3 / Plan 01 — atomic addPerson (D-09).
-- Inserts a new person and patches the anchor's symmetric relation array
-- in a single transaction. Cycle prevention (DATA-07) and parent-cap
-- (DATA-06 max 2) are enforced by the existing CHECK constraint and
-- creates_parent_cycle trigger; the RPC inherits both. PostgREST surfaces
-- those failures as Postgres errors which the Server Action wraps as
-- `addPerson failed: <message>` per threat model T-03-01.

CREATE OR REPLACE FUNCTION public.add_person_with_relation(
  p_tree_id  uuid,
  p_kind     text,
  p_anchor_id uuid,
  p_pos_x    numeric,
  p_pos_y    numeric
) RETURNS public.people
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_anchor public.people;
  v_new_id uuid := gen_random_uuid();
  v_uid    text := auth.jwt() ->> 'sub';
  v_new    public.people;
BEGIN
  -- T-03-03 (Elevation of Privilege): explicit membership check before
  -- bypassing RLS. SECURITY DEFINER strips the caller's role; we re-derive
  -- the Clerk userId from the JWT and require an active tree_members row
  -- with role IN ('owner','editor').
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'unauthenticated';
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM public.tree_members
    WHERE tree_id = p_tree_id
      AND user_id = v_uid
      AND status  = 'active'
      AND role IN ('owner', 'editor')
  ) THEN
    RAISE EXCEPTION 'forbidden';
  END IF;

  IF p_kind NOT IN ('parent', 'spouse', 'child', 'sibling') THEN
    RAISE EXCEPTION 'invalid kind: %', p_kind;
  END IF;

  SELECT * INTO v_anchor
    FROM public.people
   WHERE id = p_anchor_id AND tree_id = p_tree_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'anchor not found';
  END IF;

  INSERT INTO public.people (
    id, tree_id, name, gender, x, y, is_me,
    parent_ids, spouse_ids, child_ids
  ) VALUES (
    v_new_id, p_tree_id, '', 'u', p_pos_x, p_pos_y, false,
    CASE WHEN p_kind = 'child'   THEN ARRAY[p_anchor_id]::uuid[]
         WHEN p_kind = 'sibling' THEN COALESCE(v_anchor.parent_ids, '{}'::uuid[])
         ELSE '{}'::uuid[] END,
    CASE WHEN p_kind = 'spouse'  THEN ARRAY[p_anchor_id]::uuid[]
         ELSE '{}'::uuid[] END,
    CASE WHEN p_kind = 'parent'  THEN ARRAY[p_anchor_id]::uuid[]
         ELSE '{}'::uuid[] END
  ) RETURNING * INTO v_new;

  -- Symmetric anchor patch — exactly one of the four arrays gets array_append.
  -- Sibling case requires a triangular patch: the new person inherits the
  -- anchor's parent_ids (above), and EACH of those parents must add the
  -- new child to their child_ids. Without this, the sibling shows up as
  -- a sibling in the anchor's parents' eyes only via a re-derive.
  UPDATE public.people SET
    parent_ids = CASE WHEN p_kind = 'parent' THEN array_append(parent_ids, v_new_id) ELSE parent_ids END,
    spouse_ids = CASE WHEN p_kind = 'spouse' THEN array_append(spouse_ids, v_new_id) ELSE spouse_ids END,
    child_ids  = CASE WHEN p_kind = 'child'  THEN array_append(child_ids,  v_new_id) ELSE child_ids  END
  WHERE id = p_anchor_id AND tree_id = p_tree_id;

  IF p_kind = 'sibling' AND array_length(v_anchor.parent_ids, 1) IS NOT NULL THEN
    UPDATE public.people
       SET child_ids = array_append(child_ids, v_new_id)
     WHERE id = ANY(v_anchor.parent_ids) AND tree_id = p_tree_id;
  END IF;

  RETURN v_new;
END;
$$;

GRANT EXECUTE ON FUNCTION public.add_person_with_relation(uuid, text, uuid, numeric, numeric) TO authenticated;
```

```typescript
}> {
```

```typescript
'use client';

import { useEffect, useRef, type ReactNode } from 'react';

export type ModalProps = {
  /** Width in px — search palette uses 520 (UI-SPEC §5); default 480 from handoff. */
  width?: number;
  /**
   * Top offset in px from viewport top. When null/undefined, modal is
   * vertically centered (handoff default — Phase 5 Share modal will use this).
   * Search palette in Plan 04 passes 120.
   */
  topOffset?: number | null;
  /** Required by ARIA — wraps the modal box as `aria-label`. */
  ariaLabel: string;
  /** Called when user dismisses via Esc or backdrop click. */
  onDismiss: () => void;
  children: ReactNode;
};

/**
 * Generic centered modal primitive. Backdrop dim + box + popIn keyframes
 * inherited from handoff .modal / .modal-backdrop tokens. Restores focus
 * to whatever element held it before mount on dismiss (Pitfall #10).
 *
 * Phase 3 consumer: SearchPalette (Plan 04) — width=520, topOffset=120.
 * Phase 5 consumer: Share modal (Phase 5) — width=520, topOffset=null.
 */
export function Modal({
  width = 480,
  topOffset = null,
  ariaLabel,
  onDismiss,
  children,
}: ModalProps) {
  const restoreRef = useRef<HTMLElement | null>(null);

  // Capture pre-mount focus and restore on unmount (Pitfall #10).
  useEffect(() => {
    restoreRef.current = document.activeElement as HTMLElement | null;
    return () => {
      restoreRef.current?.focus?.();
    };
  }, []);

  // Esc dismisses regardless of focus location (the modal itself owns focus
  // when open, but a defensive global listener handles edge cases).
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== 'Escape') return;
      e.stopPropagation();
      onDismiss();
    };
    window.addEventListener('keydown', onKey);
    return () => {
      window.removeEventListener('keydown', onKey);
    };
  }, [onDismiss]);

  const onBackdropMouseDown = (e: React.MouseEvent) => {
    // Click outside the box dismisses; clicks on the box are captured by
    // the box's onMouseDown stopPropagation below.
    if (e.target === e.currentTarget) onDismiss();
  };

  return (
    <div
      onMouseDown={onBackdropMouseDown}
      className="fixed inset-0 z-[100] grid"
      style={{
        background: 'oklch(0.18 0.008 80 / 0.4)',
        animation: 'fadeIn 0.15s ease',
        // When topOffset is provided, top-align the modal by switching
        // place-items from center to start. Search palette uses 120.
        placeItems: topOffset == null ? 'center' : 'start center',
      }}
    >
      <div
        role="dialog"
        aria-modal="true"
        aria-label={ariaLabel}
        onMouseDown={(e) => e.stopPropagation()}
        className="flex flex-col"
        style={{
          background: 'var(--bg-card)',
          border: '1px solid var(--ink)',
          width,
          maxWidth: 'calc(100vw - 40px)',
          boxShadow: '8px 8px 0 var(--ink)',
          animation: 'popIn 0.18s cubic-bezier(0.2, 0.9, 0.3, 1.1)',
          marginTop: topOffset ?? 0,
        }}
      >
        {children}
      </div>
    </div>
  );
}
```
