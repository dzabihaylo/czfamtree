```typescript
setRadialOpenFor: (personId) =>
  set((state) => { state.radialOpenFor = personId; }),

setAutoFocusNameOnNextMount: (v) =>
  set((state) => { state.autoFocusNameOnNextMount = v; }),

addOptimisticPerson: (newPerson, anchorId, kind) =>
  set((state) => {
    state.people[newPerson.id] = newPerson;
    const anchor = state.people[anchorId];
    if (!anchor) return;
    switch (kind) {
      case 'spouse':
        if (!anchor.spouseIds.includes(newPerson.id)) anchor.spouseIds.push(newPerson.id);
        break;
      case 'parent':
        if (!anchor.parentIds.includes(newPerson.id)) anchor.parentIds.push(newPerson.id);
        break;
      case 'child':
        if (!anchor.childIds.includes(newPerson.id)) anchor.childIds.push(newPerson.id);
        break;
      case 'sibling':
        // Sibling shares anchor's parents — RPC patches each parent's child_ids.
        // Mirror that locally so EdgeLayer derives the correct edges.
        for (const parentId of anchor.parentIds) {
          const parent = state.people[parentId];
          if (parent && !parent.childIds.includes(newPerson.id)) {
            parent.childIds.push(newPerson.id);
          }
        }
        break;
    }
  }),

rollbackOptimisticPerson: (newPersonId, anchorId, kind) =>
  set((state) => {
    delete state.people[newPersonId];
    const anchor = state.people[anchorId];
    if (!anchor) return;
    const drop = (arr: string[]) => {
      const i = arr.indexOf(newPersonId);
      if (i !== -1) arr.splice(i, 1);
    };
    switch (kind) {
      case 'spouse': drop(anchor.spouseIds); break;
      case 'parent': drop(anchor.parentIds); break;
      case 'child':  drop(anchor.childIds);  break;
      case 'sibling':
        for (const parentId of anchor.parentIds) {
          const parent = state.people[parentId];
          if (parent) drop(parent.childIds);
        }
        break;
    }
    // Drop any orphaned save-state entry from Phase 2 SAVE-04 surface.
    delete state.saveStateByPersonId[newPersonId];
  }),
```

```typescript
'use client';

import { useEffect, useRef } from 'react';
import { useTreeStore, useTreeStoreApi } from '@/lib/store/tree-store';
import { addPerson } from '@/app/actions/people';
import { initialOffsetFor, nudgePosition, type RelativeKind } from '@/lib/graph/placement';
import { NODE_W, NODE_H } from '@/lib/graph/edges';
import { cn } from '@/lib/utils/cn';

const ITEMS: ReadonlyArray<{ kind: RelativeKind; label: string; symbol: string; angleDeg: number }> = [
  { kind: 'parent',  label: 'PARENT',  symbol: '↑', angleDeg: -90 },
  { kind: 'spouse',  label: 'SPOUSE',  symbol: '↔', angleDeg: 0 },
  { kind: 'child',   label: 'CHILD',   symbol: '↓', angleDeg: 90 },
  { kind: 'sibling', label: 'SIBLING', symbol: '←', angleDeg: 180 },
];
const RADIUS = 90; // UI-SPEC §"Radial button radius"

export function RadialMenu({ treeId }: { treeId: string }) {
  const radialOpenFor = useTreeStore((s) => s.radialOpenFor);
  const setRadialOpenFor = useTreeStore((s) => s.setRadialOpenFor);
  const setSelectedPersonId = useTreeStore((s) => s.setSelectedPersonId);
  const setSidePanelOpen = useTreeStore((s) => s.setSidePanelOpen);
  const setAutoFocusNameOnNextMount = useTreeStore((s) => s.setAutoFocusNameOnNextMount);
  const addOptimisticPerson = useTreeStore((s) => s.addOptimisticPerson);
  const rollbackOptimisticPerson = useTreeStore((s) => s.rollbackOptimisticPerson);
  const setSaveState = useTreeStore((s) => s.setSaveState);
  const storeApi = useTreeStoreApi();

  // Save the anchor's `+` button HTMLElement for focus restore on Esc (Pitfall #10).
  // PersonNode passes the button via a window-level ref instead of prop drilling
  // — captured by reading document.activeElement at the moment the radial mounts.
  const focusReturnRef = useRef<HTMLElement | null>(null);
  useEffect(() => {
    if (radialOpenFor) {
      focusReturnRef.current = document.activeElement as HTMLElement | null;
    }
  }, [radialOpenFor]);

  // Esc dismisses; do NOT bail in inputs (radial would never be open with input focus,
  // but defensive — the radial supersedes any global Esc listener via stopPropagation).
  useEffect(() => {
    if (!radialOpenFor) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== 'Escape') return;
      e.stopPropagation();
      setRadialOpenFor(null);
      focusReturnRef.current?.focus?.();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [radialOpenFor, setRadialOpenFor]);

  if (!radialOpenFor) return null;
  const anchor = storeApi.getState().people[radialOpenFor];
  if (!anchor) {
    // Anchor was deleted while radial was open — silently dismiss.
    setRadialOpenFor(null);
    return null;
  }

  const handlePick = async (kind: RelativeKind) => {
    setRadialOpenFor(null);
    const newId = crypto.randomUUID();
    const initialPos = initialOffsetFor(kind, { x: anchor.x, y: anchor.y });
    const finalPos = nudgePosition(initialPos, storeApi.getState().people);

    // Optimistic-first per D-12 + RESEARCH Pattern 2.
    addOptimisticPerson(
      {
        id: newId,
        name: '',
        gender: 'u',
        pronouns: null,
        birthYear: null,
        deathYear: null,
        birthPlace: null,
        notes: null,
        spouseIds: kind === 'spouse' ? [anchor.id] : [],
        parentIds: kind === 'child' ? [anchor.id] : kind === 'sibling' ? [...anchor.parentIds] : [],
        childIds:  kind === 'parent' ? [anchor.id] : [],
        x: finalPos.x,
        y: finalPos.y,
        isMe: false,
      },
      anchor.id,
      kind,
    );
    setSelectedPersonId(newId);
    setSidePanelOpen(true);
    setAutoFocusNameOnNextMount(true);
    setSaveState(newId, 'saving');

    try {
      await addPerson(treeId, kind, anchor.id, finalPos);
      setSaveState(newId, 'saved');
      setTimeout(() => {
        if (storeApi.getState().saveStateByPersonId[newId] === 'saved') {
          setSaveState(newId, 'idle');
        }
      }, 1400);
      // NOTE: confirmation toast `Added {kind} · {anchor.name}` is published
      // by Plan 03 via pushToast(). Plan 02 ships only the success-pill flip.
    } catch (err) {
      // Rollback per D-12.
      rollbackOptimisticPerson(newId, anchor.id, kind);
      setSelectedPersonId(anchor.id);
      // SaveErrorToast (Phase 2 component, still present until Plan 03's
      // refactor) auto-surfaces because we flip the anchor's saveState to
      // 'error'. Generic toast publication via pushToast lands in Plan 03.
      setSaveState(anchor.id, 'error');
      // Returning the anchor's pill to error keeps the existing
      // SaveErrorToast.tsx subscription firing the per-person toast — copy
      // is generic ("Couldn't save changes for {name}"), Plan 03 specialises.
    }
  };

  // Anchor center — `+` button is at PersonNode bottom-center, 14px below.
  const anchorCenterX = anchor.x + NODE_W / 2;
  const anchorCenterY = anchor.y + NODE_H + 14;

  return (
    <div
      role="menu"
      aria-label={`Add relative for ${anchor.name || 'Unnamed'}`}
      className="absolute"
      style={{
        left: anchorCenterX,
        top: anchorCenterY,
        width: 0,
        height: 0,
        pointerEvents: 'none', // container itself is non-interactive; buttons accept clicks
        zIndex: 60,
      }}
    >
      {ITEMS.map((it) => {
        const rad = (it.angleDeg * Math.PI) / 180;
        const dx = Math.cos(rad) * RADIUS;
        const dy = Math.sin(rad) * RADIUS;
        return (
          <button
            key={it.kind}
            type="button"
            role="menuitem"
            aria-label={`Add ${it.kind} for ${anchor.name || 'Unnamed'}`}
            onMouseDown={(e) => e.stopPropagation()}
            onClick={(e) => {
              e.stopPropagation();
              void handlePick(it.kind);
            }}
            className={cn(
```
