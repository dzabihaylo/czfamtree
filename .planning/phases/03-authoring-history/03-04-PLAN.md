Single hook mounted at TreeCanvas root. Per RESEARCH §Pattern 3, verbatim:

```typescript
'use client';

import { useEffect } from 'react';
import { useTreeStore, useTreeStoreApi } from '@/lib/store/tree-store';

/**
 * Single keydown listener with `document.activeElement` scope gate.
 *
 * Skips when focus is in `<input>`, `<textarea>`, or `[contenteditable=true]`
 * so native input behaviors (textarea undo, browser Find-in-page) run unimpeded.
 *
 * Dispatches:
 *  - ⌘Z / Ctrl+Z         → temporal.undo()       (HIST-01, HIST-04)
 *  - ⌘⇧Z / ⌘Y / Ctrl variants → temporal.redo()
 *  - ⌘K / ⌘F             → setSearchOpen(true)   (SRCH-01)
 *
 * ⌘F calls e.preventDefault() to suppress browser Find-in-page when canvas owns focus
 * (UI-SPEC §8). All others call e.preventDefault() to consume the keystroke.
 */
export function useTreeKeyboard({ treeId: _treeId }: { treeId: string }) {
  const setSearchOpen = useTreeStore((s) => s.setSearchOpen);
  const storeApi = useTreeStoreApi();

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      // Scope gate (HIST-04 + A11Y-02 + D-08).
      const ae = document.activeElement;
      const tag = ae?.tagName;
      const isContentEditable = (ae as HTMLElement | null)?.isContentEditable ?? false;
      if (tag === 'INPUT' || tag === 'TEXTAREA' || isContentEditable) return;

      const meta = e.metaKey || e.ctrlKey;
      if (!meta) return;

      const key = e.key.toLowerCase();

      if (key === 'z') {
        e.preventDefault();
        if (e.shiftKey) storeApi.temporal.getState().redo();
        else storeApi.temporal.getState().undo();
        return;
      }
      if (key === 'y') {
        e.preventDefault();
        storeApi.temporal.getState().redo();
        return;
      }
      if (key === 'k' || key === 'f') {
        e.preventDefault(); // suppress browser Find-in-page when canvas owns focus (UI-SPEC §8)
        setSearchOpen(true);
        return;
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [storeApi, setSearchOpen]);
}
```

**Step 0 — components/ui/Modal.tsx (MODIFY) — small-viewport top clamp (RESEARCH Q4 / D-37)**

Open `components/ui/Modal.tsx` (shipped in Plan 01 Task 3) and replace the current `marginTop: topOffset ?? 0` line so the modal's top edge clamps on short viewports. This resolves RESEARCH §"Open Questions" Q4 (Modal `top: 120px` on small viewports) without deferring it to v2.

Replacement (one-line CSS expression — no API change):

```typescript
// Before:
//   marginTop: topOffset ?? 0,
// After:
//   When topOffset is null, modal is vertically centered (handoff default — no marginTop).
//   When topOffset is provided (search palette passes 120), CLAMP the rendered top so a
//   ~600px-tall viewport doesn't push the modal below the fold. The clamp expression is
//   `clamp(20px, 15vh, 120px)` — at vh=600 → 90px; at vh=800 → 120px (caller's value).
//   Keeps tall viewports identical, prevents partial-clip on short ones.
marginTop: topOffset == null ? 0 : `clamp(20px, 15vh, ${topOffset}px)`,
```

The clamp's upper bound interpolates `topOffset` so any caller that passes a non-default value (e.g., a future caller passing `topOffset={80}`) still gets the same clamp shape. SearchPalette passes `topOffset={120}` (D-20) — the clamp expression becomes literally `clamp(20px, 15vh, 120px)`.

Per UI-SPEC §Motion, the existing global `prefers-reduced-motion` rule has no effect here (clamp is a layout property, not animation).

**Step 3 — useHistoryReplay remove-person inverse case (open issue from Plan 03)**

Per Plan 03 SUMMARY's open issue, the `useHistoryReplay` diff loop in `lib/hooks/useSaveErrorToast.ts` doesn't yet handle the "person existed before, gone now" undo branch (i.e., redo of a remove).     // adjust accordingly. The radial container has role="menu".
    await page.click('[data-node][data-isme="true"] button:nth-of-type(1)'); // + button
    const radial = page.locator('[role="menu"][aria-label*="Add relative for"]');
    await expect(radial).toBeVisible();

    // 5. Click Parent slice
    await page.click('[role="menuitem"][aria-label*="Add parent for"]');

    // 6. New parent appears + side panel opens with Name field focused
    const newParent = page.locator('[data-node]:not([data-isme="true"])').first();
    await expect(newParent).toBeVisible();
    const nameInput = page.locator('input[aria-label*="Name"]'); // verify the actual aria-label in FieldInput
    await expect(nameInput).toBeFocused();

    // 7. Type a name, blur
    await nameInput.fill('Mom');
    await page.keyboard.press('Tab');

    // 8. Wait for save pill green
    await expect(page.locator('[data-savepill="saved"]')).toBeVisible({ timeout: 5000 });

    // 9. Open radial again on YOU, click Child
    await seedNode.click();
    await page.click('[data-node][data-isme="true"] button:nth-of-type(1)');
    await page.click('[role="menuitem"][aria-label*="Add child for"]');
    const newChild = page.locator('[data-node]:not([data-isme="true"])').nth(1);
    await expect(newChild).toBeVisible();
    await page.locator('input[aria-label*="Name"]').fill('Kid');
    await page.keyboard.press('Tab');

    // 10. ⌘Z — child name reverts (or whole add reverts depending on granularity)
    //    With HIST-05 + D-07 field-blur granularity, the FIRST ⌘Z reverts the
    //    Kid name write. The SECOND ⌘Z reverts the add-Child action.
    //    The demo path step 'Cmd+Z reverts each step' is interpreted as
    //    'pressing Cmd+Z multiple times unwinds the additions'.
    await page.keyboard.press('Meta+Z');
    // Either the kid name is gone OR the kid is gone — assert ONE of those is true:
    const kidNode = page.locator('[data-node]:has-text("Kid")');
    // ...

    // 11. ⌘⇧Z — redo
    await page.keyboard.press('Meta+Shift+Z');

    // 12. Drag the new parent 100px right
    const before = await newParent.boundingBox();
    if (!before) throw new Error('parent bbox not found');
    await page.mouse.move(before.x + before.width / 2, before.y + before.height / 2);
    await page.mouse.down();
    await page.mouse.move(before.x + before.width / 2 + 100, before.y + before.height / 2);
    await page.mouse.up();

    // 13. ⌘Z — drag reverts in ONE press (HIST-05 drag granularity)
    await page.keyboard.press('Meta+Z');
    const after = await newParent.boundingBox();
    if (!after) throw new Error('parent bbox after-undo not found');
    expect(Math.abs((after.x - before.x))).toBeLessThan(2);

    // 14. ⌘K — search palette opens
    await page.keyboard.press('Meta+K');
    const palette = page.locator('[role="dialog"][aria-label="Search people"]');
    await expect(palette).toBeVisible();

    // 15. Type "Mom", press Enter — recenter
    await page.locator('input[aria-label="Search people in this tree by name"]').fill('Mom');
    await page.keyboard.press('Enter');
    await expect(palette).toBeHidden();
    await expect(page.locator('[role="status"]:has-text("Centered on Mom")')).toBeVisible();

    // 16. Click Mom node (or already selected from search) → open side panel via double-click
    await page.locator('[data-node]:has-text("Mom")').dblclick();
    const sidePanel = page.locator('aside[aria-label*="person"]'); // or whatever Phase 2 uses
    await expect(sidePanel).toBeVisible();

    // 17. Click Remove — expect inline-undo toast
    await page.click('button[aria-label*="Remove"]');
    await expect(page.locator('[role="status"]:has-text("Removed Mom")')).toBeVisible();
    await expect(page.locator('[role="status"]:has-text("Removed Mom") button:has-text("Undo")')).toBeVisible();

    // 18. Click toast Undo
    await page.click('[role="status"]:has-text("Removed Mom") button:has-text("Undo")');
    await expect(page.locator('[data-node]:has-text("Mom")')).toBeVisible();
  });
});
