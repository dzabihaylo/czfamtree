**Step 2 — lib/hooks/useSaveErrorToast.ts (NEW)**

Per PATTERNS §"useSaveErrorToast.ts (NEW)" — port the existing SaveErrorToast.tsx subscription logic into a publisher hook that calls `pushToast` instead of rendering its own toast. Mounted at TreeCanvas root.

```typescript
'use client';

import { useEffect, useMemo, useRef } from 'react';
import { useTreeStore } from '@/lib/store/tree-store';
import type { SaveQueue } from '@/lib/hooks/useSaveQueue';

/**
 * Publisher hook — translates Phase 2 saveStateByPersonId 'error' transitions
 * into pushToast({ kind: 'error', dwellMs: 4400, ariaRole: 'alert' }) calls
 * on the new generic ToastHost. Replaces the bespoke <SaveErrorToast> component
 * (D-15). One toast per oldest-errored person; merges errors at the publisher
 * level per UI-SPEC §"Generic Toast Inventory" Save error row.
 */
export function useSaveErrorToast({ queue }: { queue: SaveQueue }) {
  const saveStateByPersonId = useTreeStore((s) => s.saveStateByPersonId);
  const peopleRecord = useTreeStore((s) => s.people);
  const pushToast = useTreeStore((s) => s.pushToast);

  // Find the oldest-errored person (Phase 2 SaveErrorToast convention — first
  // entry in the order Object.entries returns; relies on JS object insertion order).
  const errorPersonId = useMemo(() => {
    for (const [id, state] of Object.entries(saveStateByPersonId)) {
      if (state === 'error') return id;
    }
    return null;
  }, [saveStateByPersonId]);

  const lastPushedRef = useRef<string | null>(null);

  useEffect(() => {
    if (!errorPersonId) {
      lastPushedRef.current = null; // allow re-firing for the SAME person
                                    // after a successful retry-then-fail loop
      return;
    }
    if (lastPushedRef.current === errorPersonId) return;
    const person = peopleRecord[errorPersonId];
    if (!person) return;
    const displayName = person.name || 'Unnamed';
    pushToast({
      kind: 'error',
      message: `Couldn't save changes for ${displayName}`,
      action: {
        label: 'Retry',
        onAction: () => queue.retry(errorPersonId),
      },
      dwellMs: 4400, // D-16
      ariaRole: 'alert',
      ariaLive: 'assertive',
      personId: errorPersonId,
    });
    lastPushedRef.current = errorPersonId;
  }, [errorPersonId, peopleRecord, pushToast, queue]);
}
```

```typescript
 * Subscribe to zundo's setOnSave callback. After every undo/redo, diff prior
 * vs current `people` and fire inverse Server Actions through the per-person
 * save queue (D-06). On any inverse-call rejection, surface ONE toast
 * `Couldn't sync history` per D-04 — the local zundo state still stays
 * reverted; the UX expectation is "your local view follows ⌘Z, the server
 * may need a Retry."
 */
export function useHistoryReplay({ treeId, queue }: { treeId: string; queue: SaveQueue }) {
  const storeApi = useTreeStoreApi();
  const setSaveState = useTreeStore((s) => s.setSaveState);
  const pushToast = useTreeStore((s) => s.pushToast);

  useEffect(() => {
    const temporal = storeApi.temporal;
    let pendingErrorToastId: string | null = null;
    const flagSyncError = () => {
      // Coalesce many per-person rejections into ONE toast (D-04).
      if (pendingErrorToastId) return;
      pendingErrorToastId = pushToast({
        kind: 'error',
        message: "Couldn't sync history",
        action: {
          label: 'Retry',
          onAction: () => {
            // Retry pulls the current store state — re-issue inverse calls.
            // The simplest implementation: call temporal.undo() then redo()
            // to re-trigger setOnSave, OR walk saveStateByPersonId for 'error'
            // entries and queue.retry each. Plan 04 will exercise this; for
            // Plan 03 we capture the function and let queue.retry handle it.
            const err = storeApi.getState().saveStateByPersonId;
            for (const [pid, st] of Object.entries(err)) {
              if (st === 'error') queue.retry(pid);
            }
          },
        },
        dwellMs: 4400,
        ariaRole: 'alert',
        ariaLive: 'assertive',
      });
      // Allow the next sync-error toast to surface after this one dismisses.
      setTimeout(() => { pendingErrorToastId = null; }, 4400);
    };

    const onSave = (pastState: { people: Record<string, Person> }, _currentState: { people: Record<string, Person> }) => {
      // pastState = the snapshot we are LEAVING (the just-undone state).
      // current store state (storeApi.getState().people) = the snapshot we MOVED TO.
      // Diff and fire inverse calls per affected person.
      const current = storeApi.getState().people;
      const prior = pastState.people;
      const allIds = new Set<string>([...Object.keys(prior), ...Object.keys(current)]);
      for (const id of allIds) {
        const before = prior[id];
        const after = current[id];
        if (before && !after) {
          // Person existed before, gone now — undo deleted them. Re-create on server.
          // The simplest inverse for "removed person" is a fresh addPerson call
          // re-introducing the relationship. For Plan 03 we route through
          // queue.enqueueInverse(id, fullPriorRecord) which the queue translates
          // to an updatePerson; if the row doesn't exist server-side, we need
          // an actual addPerson — but undo of a remove only removes the row,
          // so the inverse should re-INSERT. The PR-correct path is a dedicated
          // queue.enqueueRestore(id, fullRecord) that calls a `restorePerson`
          // Server Action. **For Plan 03 v1: emit the error toast and let
          // Plan 04 finalize this branch** — most common undo cases (drag,
          // field-edit, add-relative) work today; remove-person undo of
          // optimistic delete works because Plan 04's inline-undo Delete uses
          // temporal.undo() which the Phase 2 store re-inserts the row in
          // memory; the server reconciliation loop here can fail benignly.
          // TODO(D-03/Phase-5): server-side restore-person inverse —
          // see ../../.planning/phases/03-authoring-history/03-04-PLAN.md T-03-23
          flagSyncError();
          continue;
        }
        if (!before && after) {
          // Person didn't exist before, exists now — undo of remove or redo of add.
          // For redo of an add-relative the optimistic record is in `current`;
          // the server already has it. No call needed.
          continue;
        }
        if (before && after) {
          // Diff field-by-field. Only push fields that changed.
          const prev: Partial<Record<keyof PersonPatch, unknown>> = {};
          if (before.name !== after.name) prev.name = before.name;
          if (before.gender !== after.gender) prev.gender = before.gender;
          if (before.pronouns !== after.pronouns) prev.pronouns = before.pronouns;
          if (before.birthYear !== after.birthYear) prev.birthYear = before.birthYear;
          if (before.deathYear !== after.deathYear) prev.deathYear = before.deathYear;
          if (before.birthPlace !== after.birthPlace) prev.birthPlace = before.birthPlace;
          if (before.notes !== after.notes) prev.notes = before.notes;
          // Position diff — route through movePerson via enqueueMove (existing).
          if (before.x !== after.x || before.y !== after.y) {
            // For the inverse, restore the PREVIOUS coords (which are now in
            // `before` because we landed in the new past).
            // Actually: in zundo's setOnSave on undo, current state has the
            // PRIOR snapshot already applied. So `before` in the diff is the
            // OLD-NEW (undone-from) and current is the OLD (the restored).
            // The server should reflect the restored state, so we re-issue
            // movePerson with current.x / current.y. Phase 2 movePerson is
            // already idempotent — calling it with the just-applied coords
            // is the right inverse.
            queue.enqueueMove(id, after.x, after.y);
          }
          if (Object.keys(prev).length > 0) {
            // Note: we want the SERVER to converge on the just-restored state,
            // which is `after.*`. For an undo: `after` reflects the prior state.
            // Translate field-level patch into an explicit update with the
            // current values:
            queue.enqueueInverse(id, {
              name: after.name,
              gender: after.gender,
              pronouns: after.pronouns ?? null,
              birthYear: after.birthYear ?? null,
              deathYear: after.deathYear ?? null,
              birthPlace: after.birthPlace ?? null,
              notes: after.notes ?? null,
            } as PersonPatch);
          }
        }
      }
    };

    temporal.getState().setOnSave(onSave);
    return () => {
      // zundo doesn't expose a clearOnSave; setting an empty handler unsubscribes.
      temporal.getState().setOnSave(() => undefined);
    };
  }, [storeApi, queue, setSaveState, pushToast, treeId]);
}
```

```typescript
import { useEffect } from 'react';
import { useTreeStore, type Toast as ToastType } from '@/lib/store/tree-store';

const TOAST_HEIGHT_PX = 44; // approx — 13px text + 10*2 padding + button row
const TOAST_GAP_PX = 8;
const FIRST_TOAST_BOTTOM_PX = 80;

function Toast({ toast, index }: { toast: ToastType; index: number }) {
  const dismissToast = useTreeStore((s) => s.dismissToast);

  useEffect(() => {
    const t = setTimeout(() => dismissToast(toast.id), toast.dwellMs);
    return () => clearTimeout(t);
  }, [toast.id, toast.dwellMs, dismissToast]);

  return (
    <div
      role={toast.ariaRole}
      aria-live={toast.ariaLive}
      className="fixed flex items-center"
      style={{
        bottom: FIRST_TOAST_BOTTOM_PX + index * (TOAST_HEIGHT_PX + TOAST_GAP_PX),
        left: '50%',
        transform: 'translateX(-50%)',
        background: 'var(--ink)',
        color: 'var(--bg)',
        padding: '10px 16px',
        gap: 12,
        fontFamily: 'var(--sans)',
        fontSize: 13,
        fontWeight: 400,
        boxShadow: '3px 3px 0 var(--accent)',
        zIndex: 200,
        animation: 'toastIn 180ms ease',
      }}
    >
      <span>{toast.message}</span>
      {toast.action && (
        <button
          type="button"
          onClick={() => {
            toast.action!.onAction();
            dismissToast(toast.id);
          }}
          aria-label={toast.action.label}
          className="hover:underline"
          style={{
            color: 'var(--accent)',
            fontWeight: 600,
          }}
        >
          {toast.action.label}
        </button>
      )}
    </div>
  );
}

export function ToastHost() {
  const toasts = useTreeStore((s) => s.toasts);
  if (toasts.length === 0) return null;
  return (
    <div role="region" aria-label="Notifications">
      {toasts.map((t, i) => (
        <Toast key={t.id} toast={t} index={i} />
      ))}
    </div>
  );
}
```
